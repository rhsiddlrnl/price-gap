배달하면 얼마 더? - PHP 실습용 SQL
(Web-DB 예제 book/SQL.txt 의 테이블 1개(book)를 테이블 4개로 바꾼 것)

테이블 4개
- brand          : 브랜드          음식 종류(food_type)가 있어서 치킨 말고 피자·버거 브랜드도 추가 가능
- menu           : 메뉴와 매장가   매장가 = 공식 홈페이지 권장소비자가격
- platform       : 배달앱          default_markup = 조사 전 기본 배달가 규칙 (매장가 + 이 금액)
- delivery_price : 메뉴 x 배달앱   is_default = 1 이면 기본값, 0 이면 직접 조사한 가격

데이터 출처
- 매장가 : 교촌치킨 공식 홈페이지 치킨 메뉴 77종의 권장소비자가격 (https://www.kyochon.com/menu/chicken.asp, 2026-10-05 확인)
- 배달가 : 배달앱 API 를 쓸 수 없어서 아직 기본값입니다.
           배달의민족·쿠팡이츠·요기요 = 매장가 + 2,000원, 자사앱 = 매장가
           (근거 : 2025년 9월 서울 교촌치킨 가맹점 대부분이 배달앱 가격만 2,000원 올리고 매장·자사앱 가격은 유지했다는 보도)
           직접 조사한 가격은 price.php(배달가 입력) 화면이나 아래 6) 의 UPDATE 문으로 바꿉니다.


1) 새 데이터베이스 만들기

- phpMyAdmin 좌측상단 '새로운' 아이콘 클릭
- 데이터베이스명 PriceGapDB 입력 (리눅스 MySQL 은 대소문자를 구분하므로 정확히)
- 데이터정렬방식 utf8mb4_general_ci 입력
- '만들기' 버튼 클릭
- 아래 2) ~ 4) 의 SQL 문을 차례로 SQL 탭에 붙여넣고 실행


2) 데이터베이스 PriceGapDB 에 테이블 4개 생성

CREATE TABLE `brand` (
  `brand_id`   INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `name`       VARCHAR(50)  NOT NULL UNIQUE COMMENT '브랜드 (교촌치킨)',
  `food_type`  VARCHAR(20)  NOT NULL COMMENT '음식 종류 (치킨, 피자, 버거 ...)',
  `homepage`   VARCHAR(255) NULL COMMENT '매장가 출처 (공식 홈페이지 메뉴 페이지)',
  `checked_on` DATE         NULL COMMENT '매장가 확인일'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `platform` (
  `platform_id`    INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `name`           VARCHAR(30) NOT NULL UNIQUE COMMENT '배달앱 (배달의민족, 쿠팡이츠, 요기요, 자사앱)',
  `default_markup` INT NOT NULL DEFAULT 0 COMMENT '조사 전 기본 배달가 = 매장가 + 이 금액',
  `sort_order`     INT NOT NULL DEFAULT 0 COMMENT '화면 표시 순서'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `menu` (
  `menu_id`     INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `brand_id`    INT NOT NULL,
  `category`    VARCHAR(30)  NOT NULL COMMENT '분류 (공식 홈페이지 기준 : 허니시리즈 ...)',
  `name`        VARCHAR(100) NOT NULL COMMENT '메뉴 이름',
  `store_price` INT NOT NULL COMMENT '매장가 (공식 홈페이지 권장소비자가격)',
  `size_info`   VARCHAR(100) NULL COMMENT '양·구성 (10호 닭, 조리 전 중량 ...)',
  `image`       VARCHAR(255) NULL COMMENT '메뉴 이미지 주소',
  UNIQUE KEY `uq_menu` (`brand_id`, `name`),
  FOREIGN KEY (`brand_id`) REFERENCES `brand` (`brand_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE `delivery_price` (
  `menu_id`      INT NOT NULL,
  `platform_id`  INT NOT NULL,
  `price`        INT NOT NULL COMMENT '배달앱 메뉴 가격',
  `delivery_fee` INT NULL COMMENT '배달팁 (NULL = 모름)',
  `is_default`   BOOLEAN NOT NULL DEFAULT TRUE COMMENT '1 = 기본값(조사 전), 0 = 직접 조사',
  `store_name`   VARCHAR(50) NULL COMMENT '조사한 매장 (예: 교촌치킨 OO점)',
  `checked_on`   DATE NULL COMMENT '조사일',
  PRIMARY KEY (`menu_id`, `platform_id`),
  FOREIGN KEY (`menu_id`) REFERENCES `menu` (`menu_id`) ON DELETE CASCADE,
  FOREIGN KEY (`platform_id`) REFERENCES `platform` (`platform_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

4) 배달가 기본값 만들기 : 모든 메뉴 x 모든 배달앱 (77 x 4 = 308개)

INSERT IGNORE INTO `delivery_price` (`menu_id`, `platform_id`, `price`, `is_default`)
SELECT m.`menu_id`, p.`platform_id`, m.`store_price` + p.`default_markup`, TRUE
FROM `menu` m CROSS JOIN `platform` p;

※ INSERT IGNORE : 이미 있는 (메뉴, 배달앱) 조합은 건너뜀
   → 메뉴나 배달앱을 SQL 로 추가한 뒤 이 문장을 다시 실행하면 빠진 기본값만 채워짐 (직접 조사한 값은 그대로)
